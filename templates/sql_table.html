🗄️ Структура базы данных

SQL-код для создания всех таблиц проекта

📁 Таблица: categories (категории эмоций)

CREATE TABLE categories (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL UNIQUE,
    icon VARCHAR(50) NOT NULL DEFAULT '📍',
    color VARCHAR(7) NOT NULL DEFAULT '#FF6B6B',
    is_active BOOLEAN DEFAULT 1
);

📍 Таблица: places (места на карте)

CREATE TABLE places (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(200) NOT NULL,
    category_id INTEGER NOT NULL,
    description TEXT NOT NULL,
    latitude FLOAT NOT NULL,
    longitude FLOAT NOT NULL,
    address VARCHAR(300),
    photo_url VARCHAR(500),
    photo_source VARCHAR(200),
    photo_author VARCHAR(200),
    track_url VARCHAR(500),
    track_author VARCHAR(200),
    track_author_death_year INTEGER,
    track_title VARCHAR(200),
    wifi_free BOOLEAN DEFAULT 0,
    wheelchair_accessible BOOLEAN DEFAULT 0,
    age_group VARCHAR(50),
    is_verified BOOLEAN DEFAULT 0,
    votes_count INTEGER DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    views_count INTEGER DEFAULT 0,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE RESTRICT
);

💡 Таблица: suggested_places (предложенные места)

CREATE TABLE suggested_places (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(200) NOT NULL,
    category_id INTEGER NOT NULL,
    description TEXT NOT NULL,
    address VARCHAR(300),
    latitude FLOAT,
    longitude FLOAT,
    suggester_name VARCHAR(100),
    age_group VARCHAR(50),
    photo_url VARCHAR(500),
    photo_source VARCHAR(200),
    photo_author VARCHAR(200),
    track_url VARCHAR(500),
    track_author VARCHAR(200),
    track_author_death_year INTEGER,
    track_title VARCHAR(200),
    wifi_free BOOLEAN DEFAULT 0,
    wheelchair_accessible BOOLEAN DEFAULT 0,
    status VARCHAR(20) DEFAULT 'pending',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE RESTRICT
);

👑 Таблица: admins (администраторы)

CREATE TABLE admins (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username VARCHAR(80) NOT NULL UNIQUE,
    email VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(200) NOT NULL,
    full_name VARCHAR(200),
    is_super_admin BOOLEAN DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    last_login DATETIME
);

👍 Таблица: votes (голоса пользователей)

CREATE TABLE votes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    place_id INTEGER NOT NULL,
    ip_address VARCHAR(45) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (place_id) REFERENCES places(id) ON DELETE CASCADE,
    UNIQUE(place_id, ip_address)
);

🔍 Индексы для производительности

-- Индексы для таблицы places
CREATE INDEX idx_places_category_id ON places(category_id);
CREATE INDEX idx_places_coordinates ON places(latitude, longitude);
CREATE INDEX idx_places_verified ON places(is_verified);
CREATE INDEX idx_places_votes ON places(votes_count DESC);

-- Индексы для таблицы suggested_places
CREATE INDEX idx_suggested_category ON suggested_places(category_id);
CREATE INDEX idx_suggested_status ON suggested_places(status);
CREATE INDEX idx_suggested_age ON suggested_places(age_group);

-- Индексы для таблицы votes
CREATE INDEX idx_votes_place ON votes(place_id);
CREATE INDEX idx_votes_ip ON votes(ip_address);
